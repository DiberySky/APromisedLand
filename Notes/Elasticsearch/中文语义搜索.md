## 一、整体调整概览

| 模块                | 现有状况                                            | 新增/修改内容                                                                 |
|---------------------|-----------------------------------------------------|-------------------------------------------------------------------------------|
| Elasticsearch 容器  | 默认无中文分词插件                                  | 需安装 IK 分词器（便于混合搜索），并增加向量索引配置。                          |

```csharp
public class OpenAiEmbeddingGenerator : IEmbeddingGenerator
{
    private readonly HttpClient _http;
    private readonly string _apiKey;
    private readonly string _model;

    public OpenAiEmbeddingGenerator(HttpClient http, IConfiguration config)
    {
        _http = http;
        _apiKey = config["OpenAI:ApiKey"];
        _model = config["OpenAI:EmbeddingModel"] ?? "text-embedding-3-small";
    }

    public async Task<float[]> GenerateAsync(string text)
    {
        var request = new { input = text, model = _model };
        var response = await _http.PostAsJsonAsync("https://api.openai.com/v1/embeddings", request);
        var result = await response.Content.ReadFromJsonAsync<EmbeddingResponse>();
        return result.Data[0].Embedding;
    }
}
```

### 4. 升级索引映射（添加向量字段）
创建 `IndexInitializer.cs` 类，在应用启动时自动创建 `questions` 索引（如果不存在），并设置分析器和向量映射：

```csharp
using Elastic.Clients.Elasticsearch;
using Elastic.Clients.Elasticsearch.IndexManagement;
using ElasticsearchService.Models;

public static class IndexInitializer
{
    public static async Task EnsureIndexAsync(ElasticsearchClient client)
    {
        var existsResponse = await client.Indices.ExistsAsync("questions");
        if (existsResponse.Exists) return;

        var createResponse = await client.Indices.CreateAsync("questions", c => c
            .Settings(s => s
                .Analysis(a => a
                    .Analyzers(an => an
                        .Custom("ik_analyzer", ca => ca
                            .Tokenizer("ik_max_word")
                            .Filters("lowercase")
                        )
                    )
                )
            )
            .Mappings(m => m
                .Properties<ElasticQuestion>(p => p
                    .Text(t => t.Title, t => t.Analyzer("ik_analyzer"))
                    .Text(t => t.Content, t => t.Analyzer("ik_analyzer"))
                    .Keyword(t => t.Tags, k => k.IgnoreAbove(256))
                    .Date(t => t.CreatedAt)
                    .Boolean(t => t.HasAcceptedAnswer)
                    .IntegerNumber(t => t.AnswerCount)
                    // 新增向量字段，维度 384
                    .DenseVector(v => v.TitleVector, d => d.Dims(384).Index(true).Similarity(DenseVectorSimilarity.Cosine))
                    .DenseVector(v => v.ContentVector, d => d.Dims(384).Index(true).Similarity(DenseVectorSimilarity.Cosine))
                )
            )
        );

        if (!createResponse.IsValidResponse)
            throw new Exception($"创建索引失败: {createResponse.DebugInformation}");
    }
}
```

在 `Program.cs` 中启动后调用：
```csharp
var app = builder.Build();
// ... 管道配置
using (var scope = app.Services.CreateScope())
{
    var client = scope.ServiceProvider.GetRequiredService<ElasticsearchClient>();
    await IndexInitializer.EnsureIndexAsync(client);
}
app.Run();
```

### 6. 修改消息处理器，生成并存入向量
在 `QuestionCreatedHandler` 和 `QuestionUpdatedHandler` 中注入 `IEmbeddingGenerator` 并调用：

```csharp
public class QuestionCreatedHandler
{
    private readonly IEmbeddingGenerator _embedder;

    public QuestionCreatedHandler(IEmbeddingGenerator embedder)
    {
        _embedder = embedder;
    }

    public async Task HandleAsync(QuestionCreated message, ElasticsearchClient client)
    {
        var doc = new ElasticQuestion
        {
            Id = message.QuestionId,
            Title = message.Title,
            Content = StripHtml(message.Content),
            CreatedAt = message.Created,
            Tags = message.Tags,
            // 生成向量（若文本过长，可截断至模型最大长度）
            TitleVector = await _embedder.GenerateAsync(message.Title),
            ContentVector = await _embedder.GenerateAsync(StripHtml(message.Content))
        };

        var response = await client.IndexAsync(doc, idx => idx.Index("questions").Id(message.QuestionId));
        // ... 日志
    }
}
```

> **注意**：Wolverine 的消息处理器默认使用构造函数注入，需在 `Program.cs` 中启用服务定位以确保 `IEmbeddingGenerator` 可被解析（您已配置 `AlwaysUseServiceLocationFor<ElasticsearchClient>()`，同样可添加对 `IEmbeddingGenerator` 的支持，或直接使用 `IMessageContext` 的 `ServiceProvider`）。

### 7. 新增语义搜索 API
在 `ElasticController` 中添加新方法（或新建 `SemanticController`）：

```csharp
[HttpGet("semantic-search")]
public async Task<IActionResult> SemanticSearch([FromQuery] string q, [FromQuery] int size = 10)
{
    if (string.IsNullOrWhiteSpace(q))
        return BadRequest("查询词不能为空");

    // 生成查询向量
    var queryVector = await _embedder.GenerateAsync(q);
    
    var searchRequest = new SearchRequest<ElasticQuestion>("questions")
    {
        Size = size,
        Query = new KnnQuery
        {
            Field = "titleVector",       // 可同时搜索 titleVector 和 contentVector，使用多个 knn 或 script_score
            QueryVector = queryVector,
            K = size * 2,
            NumCandidates = 100
        }
        // 可与 bool 查询结合做混合搜索（参考 Elasticsearch 8.x 的 knn + query）
    };

    try
    {
        var response = await client.SearchAsync<ElasticQuestion>(searchRequest);
        if (!response.IsValidResponse)
            return Problem("语义搜索失败", response.DebugInformation);
        return Ok(response.Documents);
    }
    catch (Exception ex)
    {
        return Problem("语义搜索异常", ex.Message);
    }
}
```

**混合搜索建议**（兼顾关键词和语义）：
```csharp
var boolQuery = new BoolQuery
{
    Should = new List<Query>
    {
        new MultiMatchQuery { Fields = "title,content", Query = q, Boost = 0.5 },
        new KnnQuery { Field = "titleVector", QueryVector = queryVector, K = size, Boost = 1.0 }
    }
};
```
> 注意：`KnnQuery` 在 8.x 中可作为独立查询与 bool 组合，需确认您使用的客户端版本支持。

## 三、性能与注意事项
- **向量维度**：务必与模型输出维度一致（此处 384）。若使用其他模型（如 text-embedding-ada-002 为 1536），需相应调整映射。
- **索引效率**：开启 `index: true` 会占用内存，若数据量极大，可考虑关闭索引（仅存储），搜索时使用 `script_score` 实时计算（但性能较差）。
- **长文本处理**：多数模型有 token 限制（如 512），需对文本进行截断或分段平均池化。
- **模型部署**：若选择本地 ONNX，需将模型文件放置在容器或挂载卷中，并确保路径可读。
- **中文分词**：若未安装 IK 插件，创建索引时指定的 `ik_analyzer` 会失败，您可回退到 `standard` 或 `smartcn`，但语义搜索本身不依赖分词。
